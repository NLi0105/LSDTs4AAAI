# Turbine must have at least but only one position (geometry).
Turbine:TurbineShape
    a sh:NodeShape ;
    sh:targetClass Turbine:Turbine ;
    sh:property [
        sh:path Turbine:hasGeometry ;
        sh:minCount 1 ;
        sh:maxCount 1 ;
    ] .
