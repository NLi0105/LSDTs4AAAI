[rule1---YawToHurricaneWindDirection-Turbines:
(?t Turbine:hasCutOutWindSpeed ?cows)
(?t Turbine:hasTurbineStatus ?ts)
(?h Hurricane:hasWindSpeed ?ws)
(?h Hurricane:hasDurationAboveCutOut ?duration)
greaterThan(?duration, 0.1)
(?e Event:hasStatus "Active")
greaterThan(?ws, ?cows)
changeTo(?t Turbine:hasTurbineStatus "Shut down" ?newStatus)
changeTo(?t Turbine:hasPitchAngle "90.0"^^xsd:double ?newPitch)
->
(?t Turbine:hasTurbineStatus ?newStatus)
(?t Turbine:hasPitchAngle ?newPitch)]


[rule2---YawToHurricaneWindDirection-Turbines:
(?t Turbine:hasYawAngle ?y)
(?h Hurricane:hasWindDirection ?hwd)
(?h Hurricane:hasDurationAboveCutOut ?duration)
(?h Hurricane:hasWindDirection ?hwd)
greaterThan(?duration, 0.1)
(?e Event:hasStatus "Active")
changeTo(?t Turbine:hasYawAngle ?hwd ?newStatus)
->
(?t Turbine:hasYawAngle ?newStatus)]
