@prefix WindSpeed: <http://www.cee.umd.edu/Energy/MSP/WindSpeed#>.

// ------------------ Turbine positions ----------------------------
[rule1---ExclusionZones-Turbines:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?tp)
(?r <http://www.cee.umd.edu/Energy/MSP/Restricted#hasGeometry> ?ra)
isIntersect(?tp, ?ra)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#violateExclusionZone> "true")]

[rule2---EssentialFishHabitat-Turbines:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?tp)
(?e <http://www.cee.umd.edu/Energy/MSP/EFH#hasGeometry> ?efh)
isIntersect(?tp, ?efh)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#InEFH> "true")]

[rule3---FindWindZone-Turbines:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?tGeom)
(?ws <http://www.cee.umd.edu/Energy/MSP/WindSpeed#hasGeometry> ?wsGeom)
isIntersect(?tGeom, ?wsGeom)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#hasWindZone> ?ws)]

[rule3.1---OperationalWindZone-Turbines:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasWindZone> ?ws)
(?t <http://www.cee.umd.edu/Energy/Turbine#hasCutInWindSpeed> ?cutIn)
(?t <http://www.cee.umd.edu/Energy/Turbine#hasCutOutWindSpeed> ?cutOut)
(?ws <http://www.cee.umd.edu/Energy/MSP/WindSpeed#hasMinWindSpeed> ?minSpeed)
(?ws <http://www.cee.umd.edu/Energy/MSP/WindSpeed#hasMaxWindSpeed> ?maxSpeed)
greaterThan(?minSpeed, ?cutIn)
lessThan(?maxSpeed, ?cutOut)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#InOperationalWindZone> "true")
]

[rule4---InRangeOfExportCables-Turbines:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?tGeom)
(?ecc <http://www.cee.umd.edu/Energy/MSP/ECC#hasGeometry> ?eccGeom)
isWithinDistance(?tGeom, ?eccGeom, 20, "km")
->
(?t <http://www.cee.umd.edu/Energy/Turbine#WithinCableRange> "true")]

[rule5---InRangeOfExportCables-Substations:
(?s Substation:hasGeometry ?sGeom)
(?ecc ECC:hasGeometry ?eccGeom)
isWithinDistance(?sGeom, ?eccGeom, 2, "km")
->
(?s Substation:WithinCableRange "true")]

[rule6---AssignFoundation:
(?t Turbine:hasGeometry ?tGeom)
(?sd SeaDepth:hasGeometry ?sdGeom)
assignFoundation(?tGeom, ?sdGeom, 60.0, ?f)
->
(?t Turbine:hasFoundation ?f)]

[rule7---ConnectionToCable-Turbines:
(?t Turbine:hasGeometry ?tGeom)
(?c Cable:hasGeometry ?cGeom)
isTouching(?tGeom, ?cGeom)
->
(?t Turbine:hasCable ?c)]



// -------------------------------------- WindFarm --------------------------------------
[rule20---WindFarm-InWindLeaseArea:
(?wf WindFarm:hasGeometry ?wfGeom)
(?wl WindLease:hasGeometry ?wlGeom)
(?wf WindFarm:hasGeometry ?wfGeom)
(?wl WindLease:hasGeometry ?wlGeom)
isIntersect(?wfGeom, ?wlGeom)
->
(?wf WindFarm:InWindLeaseArea "true")]

[rule21---CoralProtectionAreas-WindFarm:
(?wf WindFarm:hasGeometry> ?wfGeom)
(?c Coral:hasGeometry> ?cGeom)
isIntersect(?wfGeom, ?cGeom)
->
(?wf WindFarm:violateCoralProtectionArea "true")]

[rule22---EssentialFishHabitat-WindFarm:
(?wf WindFarm:hasGeometry> ?wfGeom)
(?efh EFH:hasGeometry> ?efhGeom)
isIntersect(?wfGeom, ?efhGeom)
->
(?wf WindFarm:InEFH> "true")]
