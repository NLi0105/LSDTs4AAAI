// --------------- GeoSpatial rules --------------------------------

// ------------------ Turbine positions ------------------------------
[ruleInside:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?p)
(?w <http://www.cee.umd.edu/Energy/WindFarm#hasGeometry> ?poly)
isInside(?p, ?poly)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#hasInside> "true")]

[ruleOutside:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?p)
(?w <http://www.cee.umd.edu/Energy/WindFarm#hasGeometry> ?poly)
isOutside(?p, ?poly)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#hasOutside> "true")]

[ruleIntersect:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?p)
(?w <http://www.cee.umd.edu/Energy/WindFarm#hasGeometry> ?poly)
isIntersect(?p, ?poly)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#hasIntersect> "true")]

[ruleTouch:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasGeometry> ?p)
(?w <http://www.cee.umd.edu/Energy/WindFarm#hasGeometry> ?poly)
isTouching(?p, ?poly)
->
(?t <http://www.cee.umd.edu/Energy/Turbine#hasTouching> "true")]

[ruleWithinDistance:

(?t <http://www.cee.umd.edu/Energy/Turbine#hasWithinDistance> "true")]

// --------------------- Wind Farm position --------------------------

// --------------------- Wind Resource rules -------------------------
[ruleWindFarmWindSpeed:
  (?wf <http://www.cee.umd.edu/Energy/WindFarm#hasGeometry> ?wfGeom)
  (?ws <http://www.cee.umd.edu/Energy/MSP/WindSpeed#hasGeometry> ?wsGeom)
  (?ws <http://www.cee.umd.edu/Energy/MSP/WindSpeed#hasMinWindSpeed> ?minSpeed)
  (?ws <http://www.cee.umd.edu/Energy/MSP/WindSpeed#hasMaxWindSpeed> ?maxSpeed)
  isIntersect(?wfGeom, ?wsGeom)
  ->
  (?wf <http://www.cee.umd.edu/Energy/WindFarm#hasMinWindSpeed> ?minSpeed)
  (?wf <http://www.cee.umd.edu/Energy/WindFarm#hasMaxWindSpeed> ?maxSpeed)
]

// --------------------- Triple manipulation rules -------------------

[ruleChangeValue:
(?t <http://www.cee.umd.edu/Energy/Turbine#hasTurbineStatus> ?ts)
(?wr <http://www.cee.umd.edu/Energy/WindResource#hasWindSpeed50M> ?ws50)

(?t <http://www.cee.umd.edu/Energy/Turbine#hasTurbineStatus> ?newStatus)]
